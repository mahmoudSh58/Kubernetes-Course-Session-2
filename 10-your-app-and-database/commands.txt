# --- Slides 58-59: Dockerfile -> image -> minikube nodes -> Pod (no registry) ---
docker build -t webapp-api:v1 ./10-your-app-and-database/api
minikube image ls | grep webapp-api           # not there yet
minikube image load webapp-api:v1             # copy into every node
minikube image ls | grep webapp-api           # now it is

# Generate the Deployment YAML (already saved as webapp-api.yaml, with IfNotPresent)
kubectl create deployment webapp-api --image=webapp-api:v1 \
  --dry-run=client -o yaml > /tmp/webapp-api-generated.yaml

# Replace the nginx demo Deployment from 05-labels-selectors with your own image
kubectl delete deployment webapp-api -n webapp-stack --ignore-not-found
kubectl apply -f 10-your-app-and-database/webapp-config.yaml   # adds DB_HOST=mysql-svc
kubectl apply -f 10-your-app-and-database/webapp-api.yaml
kubectl get pods -n webapp-stack -l app=webapp-api            # Running, no pull
kubectl exec -n webapp-stack deploy/webapp-api -- printenv DB_HOST
curl http://127.0.0.1/home                                    # through the Ingress (tunnel running)

# Example v1 -> v2: new tag + load + rollout
sed -i 's/v1/v2/' 10-your-app-and-database/api/index.html
docker build -t webapp-api:v2 ./10-your-app-and-database/api
minikube image load webapp-api:v2
kubectl set image deploy/webapp-api -n webapp-stack webapp-api=webapp-api:v2
kubectl rollout status deploy/webapp-api -n webapp-stack
curl http://127.0.0.1/home                                    # Hello from webapp-api v2
kubectl rollout undo deploy/webapp-api -n webapp-stack        # back to v1
sed -i 's/v2/v1/' 10-your-app-and-database/api/index.html

# ErrImagePull?  wrong tag, image not loaded, or imagePullPolicy: Always
kubectl describe pod -n webapp-stack -l app=webapp-api | grep -A5 Events


# --- Slide 60: the database Deployment ---
# mysql-secret comes from 03-configmaps-secrets:
kubectl get secret mysql-secret -n webapp-stack || \
  kubectl apply -f 03-configmaps-secrets/mysql-secret.yaml

kubectl apply -f 10-your-app-and-database/mysql-pvc.yaml
kubectl apply -f 10-your-app-and-database/mysql-deployment.yaml
kubectl apply -f 10-your-app-and-database/mysql-svc.yaml
kubectl get pvc,pods,svc,endpointslices -n webapp-stack
kubectl rollout status deploy/mysql -n webapp-stack           # MySQL needs ~30-60 s
kubectl logs deploy/mysql -n webapp-stack --tail=5            # "ready for connections"

# Write some data
kubectl exec -it -n webapp-stack deploy/mysql -- \
  sh -c 'mysql -uroot -p"$MYSQL_ROOT_PASSWORD" -e "
    CREATE DATABASE IF NOT EXISTS shop;
    CREATE TABLE IF NOT EXISTS shop.guests (name VARCHAR(50));
    INSERT INTO shop.guests VALUES (\"Session 2\");
    SELECT * FROM shop.guests;"'

# Delete the Pod live -> the new one finds the same tables (data is on the claim)
kubectl delete pod -n webapp-stack -l app=mysql
kubectl rollout status deploy/mysql -n webapp-stack
kubectl exec -it -n webapp-stack deploy/mysql -- \
  sh -c 'mysql -uroot -p"$MYSQL_ROOT_PASSWORD" -e "SELECT * FROM shop.guests;"'

# Reach the database by its Service name (DNS) from another Pod
kubectl run mysql-client -n webapp-stack --rm -it --image=mysql:8.4 --restart=Never -- \
  mysql -h mysql-svc -uroot -p'S3cret!' -e 'SHOW DATABASES;'

# THIS FIELD                          IN                     MUST MATCH                  IF IT DOESN'T
# claimName: mysql-pvc                mysql Deployment       the PVC's name              Pod stuck in ContainerCreating
# secretKeyRef: mysql-secret/DB_PASSWORD  mysql Deployment   the Secret's name and key   CreateContainerConfigError
# configMapRef: webapp-config         webapp-api Deployment  the ConfigMap's name        CreateContainerConfigError
# selector: app: mysql                mysql-svc Service      the Pod template labels     Service has no endpoints
# targetPort: 3306                    mysql-svc Service      the port MySQL listens on   Connection refused
# DB_HOST=mysql-svc                   webapp-config          the Service's name (DNS)    App cannot resolve the database
# backend.service.name                Ingress                the Service's name and port 503 from the controller
