# --- Slide 28: ConfigMap - imperative ---
kubectl create cm demo-config -n webapp-stack --from-literal=LOG_LEVEL=info
echo 'server { listen 8081; }' > default.conf
kubectl create cm nginx-config -n webapp-stack --from-file=default.conf
kubectl get cm -n webapp-stack
kubectl get cm nginx-config -n webapp-stack -o yaml
kubectl delete cm demo-config nginx-config -n webapp-stack; rm default.conf


# --- Slide 29: ConfigMap as env vars and as a file ---
kubectl apply -f 03-configmaps-secrets/webapp-config.yaml
kubectl get cm webapp-config -n webapp-stack -o yaml
kubectl apply -f 03-configmaps-secrets/nginx-config-pod.yaml
kubectl get pod nginx-config -n webapp-stack

# Verify: env var
kubectl exec -n webapp-stack nginx-config -- printenv LOG_LEVEL
#   info
# envFrom skips invalid names -> default.conf is NOT an env var, only a file
kubectl exec -n webapp-stack nginx-config -- cat /etc/nginx/conf.d/default.conf
kubectl exec -n webapp-stack nginx-config -- curl -s localhost:8081 | head -n 4

# Change the port in the ConfigMap, restart the Pod, show nginx on the new port
kubectl edit cm webapp-config -n webapp-stack          # listen 8081 -> 8082
kubectl delete pod nginx-config -n webapp-stack
kubectl apply -f 03-configmaps-secrets/nginx-config-pod.yaml
kubectl exec -n webapp-stack nginx-config -- cat /etc/nginx/conf.d/default.conf
# (set it back to 8081 afterwards - the Services chapter uses 8081)


# --- Slide 30/31: Secrets and the base64 trap ---
kubectl create secret generic mysql-secret -n webapp-stack \
  --from-literal=DB_USER=app --from-literal=DB_PASSWORD='S3cret!'
#   or declaratively:  kubectl apply -f 03-configmaps-secrets/mysql-secret.yaml

kubectl get secret mysql-secret -n webapp-stack -o yaml   # values are base64
kubectl get secret mysql-secret -n webapp-stack \
  -o jsonpath='{.data.DB_PASSWORD}' | base64 -d
#   S3cret!

# A Pod reading the Secret through envFrom: secretRef
kubectl apply -f 03-configmaps-secrets/secret-env-pod.yaml
kubectl logs -n webapp-stack secret-reader
kubectl delete pod secret-reader -n webapp-stack

# Change a Secret value (re-create in place), then restart the Pods that use it
kubectl create secret generic mysql-secret -n webapp-stack \
  --from-literal=DB_USER=app --from-literal=DB_PASSWORD='N3wS3cret!' \
  --dry-run=client -o yaml | kubectl apply -f -
# kubectl rollout restart deployment/<name> -n webapp-stack

############################################################
# Slide 32 - Configuration best practices
############################################################
# 1 Never commit real Secrets  -> Sealed Secrets, External Secrets Operator, cloud secret manager
# 2 Encrypt etcd               -> encryption at rest + RBAC on "get secrets"
# 3 Prefer files for secrets   -> env vars leak into logs, crash dumps, child processes
# 4 immutable: true            -> avoids drift, reduces API server load
# 5 Version config names       -> webapp-config-v2: clean rollout, easy rollback
# 6 One ConfigMap per app      -> same namespace as the app
