# --- Slide 70: quickest Job ever (imperative) ---
kubectl create job hello -n webapp-stack --image=busybox:1.36 -- echo hi
kubectl wait --for=condition=complete job/hello -n webapp-stack
kubectl logs job/hello -n webapp-stack
kubectl delete job hello -n webapp-stack


# --- Slide 21: a Job that runs to completion ---
kubectl apply -f 02-jobs-cronjobs/seed-job.yaml
kubectl get jobs,pods -n webapp-stack
kubectl wait --for=condition=complete job/seed-content -n webapp-stack
kubectl logs job/seed-content -n webapp-stack     # the Pod stays until you delete the Job
kubectl describe job seed-content -n webapp-stack

# Check the file it wrote on the claim
minikube ssh -- cat /mnt/data/index.html
#   <h1>Seeded</h1>

# A Job's Pod template cannot change after creation -> delete and recreate
kubectl delete job seed-content -n webapp-stack


# --- Slide 24: a CronJob that updates a file ---
kubectl apply -f 02-jobs-cronjobs/stamp-cronjob.yaml
kubectl get cj,jobs -n webapp-stack
kubectl get jobs -n webapp-stack -w          # a new Job every minute (Ctrl+C to stop)

# Test without waiting a minute
kubectl create job --from=cronjob/stamp stamp-now -n webapp-stack
kubectl logs job/stamp-now -n webapp-stack
minikube ssh -- cat /mnt/data/time.txt

# Pause / resume future runs
kubectl patch cronjob stamp -n webapp-stack -p '{"spec":{"suspend":true}}'
kubectl patch cronjob stamp -n webapp-stack -p '{"spec":{"suspend":false}}'

# Clean up
kubectl delete cronjob stamp -n webapp-stack
kubectl delete job stamp-now -n webapp-stack

# Slide 20 - Pick the controller by how long work runs
############################################################
#   Deployment  forever, replaced if it stops      restartPolicy Always          nginx, an API
#   Job         until it succeeds, then stops      Never or OnFailure            migrate a database
#   CronJob     creates a Job on a schedule        Never or OnFailure            nightly backup
#   DaemonSet   one Pod per node, forever          Always                        log/metrics agent
