############################################################
# 04 - THE NETWORKING MODEL  (Chapter 4 - slides 34-36, 73)
# Every Pod gets an IP, and every Pod can reach every other Pod.
############################################################

# Four rules (the CNI plugin implements them):
#   1 Every Pod gets its own IP           2 Pods reach Pods without NAT
#   3 Nodes reach all Pods                4 Containers in a Pod share localhost

# --- Which CNI plugin is running? (course cluster: Calico) ---
kubectl get pods -n kube-system -o wide | grep -Ei 'calico|cilium|flannel|kindnet'
kubectl get nodes -o wide                       # node IPs (e.g. 192.168.49.2)

# --- Pod IPs ---
kubectl apply -f 04-networking-model/two-pods.yaml
kubectl get pods -n webapp-stack -l demo=net -o wide     # IP + NODE columns

# --- Pod A calls Pod B directly by IP (no NAT, no Service) ---
POD_B_IP=$(kubectl get pod pod-b -n webapp-stack -o jsonpath='{.status.podIP}')
echo $POD_B_IP
kubectl exec -n webapp-stack pod-a -- curl -s http://$POD_B_IP | head -n 4

# --- Pod IPs change on every restart -> why we need Services ---
kubectl delete pod pod-b -n webapp-stack
kubectl apply -f 04-networking-model/two-pods.yaml
kubectl get pods -n webapp-stack -l demo=net -o wide     # pod-b has a NEW IP

# Clean up
kubectl delete -f 04-networking-model/two-pods.yaml
